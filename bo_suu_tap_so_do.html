Chương 2 - Mục 2.2.1 5. Sơ đồ thực thể liên kết (ERD)
Vị trí chèn: Ở Mục 2.2.1 trước các bảng CSDL chi tiết.
erDiagram
    SERVICES ||--o{ APPOINTMENTS : "được đặt trong"
    THERAPISTS ||--o{ APPOINTMENTS : "thực hiện"
    VOUCHERS ||--o{ APPOINTMENTS : "áp dụng cho"
    USERS ||--o{ APPOINTMENTS : "quản lý / đặt"

    SERVICES {
        varchar(20) id PK
        nvarchar(255) name
        nvarchar(100) category
        int duration
        decimal price
        varchar(50) icon
        nvarchar_max description
        nvarchar_max benefits
    }

    THERAPISTS {
        varchar(20) id PK
        nvarchar(100) name
        nvarchar(255) title
        varchar(50) icon
        float rating
        nvarchar(50) experience
        nvarchar_max specialty
    }

    VOUCHERS {
        varchar(50) code PK
        varchar(20) type
        int value
        decimal minOrder
        nvarchar(255) description
        bit isActive
    }

    APPOINTMENTS {
        varchar(30) id PK
        nvarchar(150) customerName
        varchar(20) customerPhone
        varchar(20) serviceId FK
        varchar(20) therapistId FK
        date appointmentDate
        varchar(10) appointmentTime
        decimal originalPrice
        varchar(50) voucherCode FK
        decimal discountAmount
        decimal finalPrice
        varchar(20) paymentMethod
        varchar(20) status
        nvarchar_max notes
        datetime createdAt
    }

    USERS {
        varchar(50) id PK
        varchar(100) username
        varchar(255) email
        varchar(20) phone
        nvarchar(150) fullName
        varchar(255) password
        varchar(20) role
        datetime createdAt
    }
